PROBLEM 1: EMPLOYEE & SALARY ANALYTICS + SIMULATING SQL 'HAVING'
Goal    : Group by department, calculate headcount, total, avg, max, min salary;
          filter departments where avg_sal > 50000 and headcount >= 2; sort DESC.
Key Ops : $match, $group, $project, $round, $sort
--------------------------------------------------------------------------------
db.emp.aggregate([
  { $group: {
      _id: "$dept",
      headcount: { $sum: 1 },
      total_sal: { $sum: "$salary" },
      avg_sal: { $avg: "$salary" },
      max_sal: { $max: "$salary" },
      min_sal: { $min: "$salary" }
  }},
  // Simulate SQL HAVING
  { $match: { avg_sal: { $gt: 50000 }, headcount: { $gte: 2 } } },
  { $project: {
      _id: 0,
      department: "$_id",
      headcount: 1,
      total_sal: 1,
      avg_sal: { $round: ["$avg_sal", 2] },
      max_sal: 1,
      min_sal: 1
  }},
  { $sort: { avg_sal: -1 } }
]);


--------------------------------------------------------------------------------
PROBLEM 2: ACADEMIC GRADING & MULTI-TIER CONDITIONAL LOGIC
Goal    : Compute total, average, Pass/Fail (avg >= 50), and multi-tier grade
          (Distinction >= 75, First Class >= 60, Pass >= 50, Fail < 50).
Key Ops : $project, $add, $divide, $cond, $switch, $toUpper
--------------------------------------------------------------------------------
db.students.aggregate([
  { $project: {
      _id: 1,
      name: { $toUpper: "$name" },
      dept: 1,
      total: { $add: ["$m1", "$m2"] },
      avg: { $divide: [{ $add: ["$m1", "$m2"] }, 2] }
  }},
  { $project: {
      _id: 1,
      name: 1,
      dept: 1,
      total: 1,
      avg: 1,
      status: {
        $cond: { if: { $gte: ["$avg", 50] }, then: "PASS", else: "FAIL" }
      },
      grade: {
        $switch: {
          branches: [
            { case: { $gte: ["$avg", 75] }, then: "DISTINCTION" },
            { case: { $gte: ["$avg", 60] }, then: "FIRST_CLASS" },
            { case: { $gte: ["$avg", 50] }, then: "PASS_CLASS" }
          ],
          default: "FAIL"
        }
      }
  }}
]);


--------------------------------------------------------------------------------
PROBLEM 3: TOP-N RECORDS PER GROUP (HIGHEST PAID IN EACH DEPARTMENT)
Goal    : Find the single highest-paid employee inside each department.
Key Ops : $sort -> $group ($push: "$$ROOT") -> $project ($slice)
--------------------------------------------------------------------------------
db.emp.aggregate([
  // 1. Sort all documents by salary descending first
  { $sort: { salary: -1 } },

  // 2. Group by dept and push ordered full documents into array
  { $group: {
      _id: "$dept",
      top_employees: { $push: "$$ROOT" }
  }},

  // 3. Slice top 1 from each group's array (change 1 to N for Top-N)
  { $project: {
      _id: 0,
      dept: "$_id",
      top_earner: { $slice: ["$top_employees", 1] }
  }},

  // 4. Unwind array to flatten output
  { $unwind: "$top_earner" }
]);

--------------------------------------------------------------------------------
PROBLEM 5: TWO-COLLECTION JOINS ($lookup - INNER & LEFT OUTER)
Goal    : Join emp with dept_info on dept == dept_code.
Key Ops : $lookup, $unwind, $ifNull
--------------------------------------------------------------------------------
// 5A. INNER JOIN (Only keeps employees whose dept matches dept_info)
db.emp.aggregate([
  { $lookup: {
      from: "dept_info",
      localField: "dept",
      foreignField: "dept_code",
      as: "dept_details"
  }},
  // Unwind without preserveNullAndEmptyArrays drops unmatched rows (Inner Join)
  { $unwind: "$dept_details" },
  { $project: {
      _id: 0,
      name: 1,
      dept: 1,
      building: "$dept_details.building",
      city: "$dept_details.city"
  }}
]);

// 5B. LEFT OUTER JOIN (Keeps all employees even if dept has no match, e.g. Sales)
db.emp.aggregate([
  { $lookup: {
      from: "dept_info",
      localField: "dept",
      foreignField: "dept_code",
      as: "dept_details"
  }},
  { $unwind: { path: "$dept_details", preserveNullAndEmptyArrays: true } },
  { $project: {
      _id: 0,
      name: 1,
      dept: 1,
      building: { $ifNull: ["$dept_details.building", "Unassigned"] },
      city: { $ifNull: ["$dept_details.city", "Unknown"] }
  }}
]);

--------------------------------------------------------------------------------

// 3. Remove "Excel" from HR employees' skills array
db.emp.updateMany(
  { dept: "HR" },
  { $pull: { skills: "Excel" } }
);

db.emp.deleteMany(
